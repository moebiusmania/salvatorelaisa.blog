{{- /* Post-processing of rendered Markdown, for the syntax markdown-it handled through plugins and Goldmark doesn't: - `::timeline{items="a, b"}` + `::` blocks (MDC style, from the Nuxt days), which Goldmark renders as a plain paragraph. They become the same markup the old Timeline component produced (_components/content/Timeline.css). - `~~text~~`: Goldmark's strikethrough is disabled in hugo.toml because it emits (styled differently by typography.css), markdown-it used . Context: the rendered HTML string. */ -}} {{- $html := . -}} {{- range findRESubmatch `

::timeline\{(.*?)\}\n::

\n?` $html -}} {{- $attrs := dict "title" "Timeline degli eventi" "description" "Alcuni eventi che ho vissuto relativi all'articolo visualizzati in ordine cronologico" -}} {{- range findRESubmatch `(\w+)="([^"]*)"` (index . 1 | htmlUnescape) -}} {{- $attrs = merge $attrs (dict (index . 1) (index . 2)) -}} {{- end -}} {{- $items := slice -}} {{- range split ($attrs.items | default "") "," -}} {{- with strings.TrimSpace . }}{{ $items = $items | append . }}{{ end -}} {{- end -}} {{- $n := len $items -}} {{- $list := "" -}} {{- range $i, $item := $items -}} {{- $text := htmlEscape $item -}} {{- $list = printf `%s
  • %s
    • ` $list (add $i 1) $n $text $text -}} {{- end -}} {{- $describedBy := cond (ne $attrs.description "") ` aria-describedby="timeline-description"` "" -}} {{- $section := printf `
      %s
    Timeline with %d items loaded
    ` (htmlEscape $attrs.title) $describedBy $n $list $n -}} {{- $html = replace $html (index . 0) $section -}} {{- end -}} {{- $html = replaceRE `~~([^~\n]+?)~~` "$1" $html -}} {{- $html | safeHTML -}}
